1. ما هو المشروع؟
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

AP Manager هو تطبيق Web محلي يعمل على Windows لإدارة أجهزة Cisco Aironet 3600
Autonomous Access Points من خلال:

  • لوحة تحكم (Dashboard) حديثة في المتصفح
  • قراءة/كتابة بيانات AP من ملف Excel
  • الاتصال بالأجهزة عبر SSH أو Telnet
  • قراءة عدد العملاء المتصلين (Dot11Radio0 فقط — 2.4GHz)
  • تغيير SSID وكلمة مرور Wi-Fi
  • إنشاء نسخ احتياطية وسجلات لكل عملية

التطبيق يعمل محليًا فقط على:
  http://127.0.0.1:8787

لا يعتمد على Cloud أو قاعدة بيانات خارجية.

3. المتطلبات
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  • Windows 10 أو Windows 11
  • Node.js 18 LTS أو أحدث
  • Microsoft Excel Desktop (للقراءة/الكتابة عبر COM)
  • PowerShell 5.1 أو أحدث
  • Posh-SSH (يُثبَّت تلقائيًا عند التثبيت)
  • كابل Ethernet للاتصال بالـAPs أثناء تغيير Wi-Fi
  • Telnet (Port 23) و/أو SSH (Port 22) مفعّل على الأجهزة

4. التثبيت (مرة واحدة)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1) انسخ مجلد AP Manager بالكامل إلى أي مسار على Windows
2) تأكد من وجود:
     - AP-Info-Manager.xlsx
     - logo.png
3) شغّل: Install-AP-Manager.bat
4) انتظر رسالة: Installation completed successfully

يقوم ملف التثبيت بـ:
  • التحقق من Node.js
  • npm install للـ frontend والـ backend
  • تثبيت Posh-SSH
  • بناء (Build) الواجهة والخادم

5. التشغيل والإيقاف
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

▶ التشغيل:
   1) شغّل Start-AP-Manager.bat
   2) يفتح المتصفح على: http://127.0.0.1:8787
   3) إذا كان السيرفر يعمل مسبقًا، لن يُنشأ نسخة ثانية

■ الإيقاف:
   1) شغّل Stop-AP-Manager.bat
   2) يُوقف السيرفر ويُنظّف Runtime/

7. ملف Excel — AP-Info-Manager.xlsx
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

الموقع: جذر مجلد AP Manager (نفس مستوى logo.png)

أهم الأعمدة:
  A  Department
  B  Access Point Name
  C  Access Point IP
  D  Access Point Mac Address
  E  Access User
  F  Access Password (= Enable Password)
  G  New WiFi SSID
  H  New WiFi Password
  K  Clients              ← يُكتب فيها العدد دائمًا

ملاحظات:
  • يمكن فتح Excel أثناء عمل الموقع
  • إذا كان Excel مفتوحًا، تُقرأ القيم الحية (حتى بدون Save)
  • كلمات المرور لا تُرسل للمتصفح — تُقرأ من Backend فقط

إضافة AP جديد:
  1) أضف صفًا جديدًا في Excel
  2) املأ Name, IP, MAC, User, Password
  3) اضغط Reload Excel في الموقع

8. قراءة عدد Clients
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

▶ لجهاز واحد:
   اضغط Get Clients في الصف

▶ لجميع الأجهزة:
   اضغط Refresh All Clients (حد أقصى 3 اتصالات متزامنة)

السلوك:
  • أثناء التحميل: "..." في Excel | 🏃‍♂️‍➡️ في الموقع
  • عند النجاح: رقم في K | 💻×N في الموقع
  • عند 0: 0️⃣
  • عند الخطأ: ERROR | ⚠️

يتم العد من Dot11Radio0 (2.4GHz) فقط — Dot11Radio1 (5GHz) مُتجاهَل.

9. تغيير SSID أو Wi-Fi Password
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1) اكتب القيمة في Excel:
     G = New WiFi SSID
     H = New WiFi Password
2) اضغط Apply Changes في الصف

قواعد:
  • SSID فارغ → لا يتغير SSID
  • Password فارغ → لا تتغير كلمة المرور
  • Password: 8–63 حرف
  • يُنشأ Backup في Backups/ قبل أي تعديل
  • يُنفَّذ write memory ويُفعَّل SSH v2 عند الحاجة
  • بعد النجاح تُمسح G/H من Excel لأسباب أمنية

⚠ لا تفصل كابل Ethernet أثناء التطبيق!

13. أشهر الأخطاء وحلولها
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

  AP لا يرد على Ping        → تحقق من الكابل والـIP
  Port 22/23 مغلق           → فعّل SSH/Telnet على AP
  Login failed              → راجع User/Password في Excel
  Excel locked              → أغلق النسخة المقفلة
  ERROR في عمود Clients     → راجع Logs/
  ENOENT Runtime json       → أعد Start-AP-Manager.bat
  npm run dev في الجذر      → استخدم Start-AP-Manager.bat (لا يوجد package.json في الجذر)
